duration_data <- read_csv(
  "US_Accidents_March23_sampled_500k.csv",

  slice_sample(n = 100000) %>%
  mutate(
    # Parse Timestamps
    Start_Time = parse_date_time(Start_Time, orders = c("ymd HMS", "ymd HM", "ymd"), tz = "UTC"),
    End_Time   = parse_date_time(End_Time, orders = c("ymd HMS", "ymd HM", "ymd"), tz = "UTC"),
    
    # Calculate Target: Crash Duration in Minutes
    Duration_mins = as.numeric(difftime(End_Time, Start_Time, units = "mins")),
    
    # Temporal Predictors
    Hour = hour(Start_Time),
    Day_of_Week = wday(Start_Time, label = TRUE),
    Is_Rush_Hour = factor(if_else(Hour %in% c(7, 8, 9, 16, 17, 18), "Yes", "No")),

    Humidity = `Humidity(%)`,
    Visibility = `Visibility(mi)`,
    Precipitation = coalesce(`Precipitation(in)`, 0),
    Distance = `Distance(mi)`,
    
    # Factors
    State = fct_lump_n(factor(State), n = 15),
    Junction = factor(Junction),
    Traffic_Signal = factor(Traffic_Signal),
    Crossing = factor(Crossing)
  ) %>%
  # Filter data entry artifacts (durations under 2 mins or over 24 hours)
  filter(Duration_mins >= 2 & Duration_mins <= 1440) %>%
  mutate(
    # Log-transform target to handle extreme skewness
    log_Duration = log(Duration_mins)
  ) %>%
  select(
    Duration_mins, log_Duration, Hour, Day_of_Week, Is_Rush_Hour, State, 
    Distance, Temperature, Humidity, Visibility, Precipitation,
    Junction, Traffic_Signal, Crossing
  ) %>%
  drop_na()

# Calculate Performance Metrics
calc_reg_metrics <- function(actual, pred, model_name) {
  rmse_val <- sqrt(mean((actual - pred)^2))
  mae_val  <- mean(abs(actual - pred))
  ss_tot   <- sum((actual - mean(actual))^2)
  ss_res   <- sum((actual - pred)^2)
  r2_val   <- 1 - (ss_res / ss_tot)
  
  tibble(
    Model = model_name,
    MAE_Minutes  = round(mae_val, 2),
    RMSE_Minutes = round(rmse_val, 2),
    R_Squared    = round(r2_val, 4)
  )
}

metrics_summary <- bind_rows(
  calc_reg_metrics(test_results$Duration_mins, test_results$pred_mins_lm, "Linear Regression"),
  calc_reg_metrics(test_results$Duration_mins, test_results$pred_mins_rf, "Random Forest")
)

# 1. Load Data & Create Binary Target
cat("Loading and processing dataset...\n")
Loading and processing dataset...
duration_binary_data <- read_csv(
  "US_Accidents_March23_sampled_500k.csv",
  show_col_types = FALSE
) %>%
  slice_sample(n = 100000) %>%
  mutate(
    # Timestamps
    Start_Time = parse_date_time(Start_Time, orders = c("ymd HMS", "ymd HM", "ymd"), tz = "UTC"),
    End_Time   = parse_date_time(End_Time, orders = c("ymd HMS", "ymd HM", "ymd"), tz = "UTC"),
    Duration_mins = as.numeric(difftime(End_Time, Start_Time, units = "mins")),
    
    # Target Variable: Binary Classification (Threshold = 120 Mins)
    Duration_Category = factor(
      if_else(Duration_mins >= 120, "Over_2_Hours", "Under_2_Hours"),
      levels = c("Under_2_Hours", "Over_2_Hours")
    ),
    
    # Features
    Hour = hour(Start_Time),
    Day_of_Week = wday(Start_Time, label = TRUE),
    Is_Rush_Hour = factor(if_else(Hour %in% c(7, 8, 9, 16, 17, 18), "Yes", "No")),
    
    Temperature = `Temperature(F)`,
    Humidity = `Humidity(%)`,
    Visibility = `Visibility(mi)`,
    Precipitation = coalesce(`Precipitation(in)`, 0),
    Distance = `Distance(mi)`,
    
    State = fct_lump_n(factor(State), n = 15),
    Junction = factor(Junction),
    Traffic_Signal = factor(Traffic_Signal),
    Crossing = factor(Crossing),
    .row_id = row_number()
  ) %>%
  filter(Duration_mins >= 2 & Duration_mins <= 1440) %>%
  select(
    Duration_Category, Hour, Day_of_Week, Is_Rush_Hour, State, 
    Distance, Temperature, Humidity, Visibility, Precipitation,
    Junction, Traffic_Signal, Crossing, .row_id
  ) %>%
  drop_na()


# 2. Stratified Train / Test Split & Downsampling
train_ids <- duration_binary_data %>%
  group_by(Duration_Category) %>%
  slice_sample(prop = 0.80) %>%
  pull(.row_id)

train_raw <- duration_binary_data %>% filter(.row_id %in% train_ids) %>% select(-.row_id)
test_set  <- duration_binary_data %>% filter(!.row_id %in% train_ids) %>% select(-.row_id)

# Downsample Training Set to 50/50 balance
over_2h_cases  <- train_raw %>% filter(Duration_Category == "Over_2_Hours")
under_2h_cases <- train_raw %>% filter(Duration_Category == "Under_2_Hours") %>% slice_sample(n = nrow(over_2h_cases))
train_balanced <- bind_rows(over_2h_cases, under_2h_cases) %>% sample_frac(1)


# 3. Fit Random Forest Classifier
cat("Training Random Forest Classifier...\n")
Training Random Forest Classifier...
fit_rf_class <- ranger(
  formula = Duration_Category ~ .,
  data = train_balanced,
  num.trees = 300,
  probability = TRUE, # Needed for ROC-AUC
  importance = "impurity",
  seed = 42
)


# 4. Predict on Test Set & Calculate Confusion Matrix
test_preds <- test_set %>%
  mutate(
    prob_Over_2H = predict(fit_rf_class, data = test_set)$predictions[, "Over_2_Hours"],
    pred_class   = factor(if_else(prob_Over_2H >= 0.50, "Over_2_Hours", "Under_2_Hours"), 
                          levels = c("Under_2_Hours", "Over_2_Hours"))
  )

# Confusion Matrix components
tp <- sum(test_preds$pred_class == "Over_2_Hours" & test_preds$Duration_Category == "Over_2_Hours")
tn <- sum(test_preds$pred_class == "Under_2_Hours" & test_preds$Duration_Category == "Under_2_Hours")
fp <- sum(test_preds$pred_class == "Over_2_Hours" & test_preds$Duration_Category == "Under_2_Hours")
fn <- sum(test_preds$pred_class == "Under_2_Hours" & test_preds$Duration_Category == "Over_2_Hours")

# Accuracy Metrics
overall_accuracy <- mean(test_preds$pred_class == test_preds$Duration_Category)
sensitivity      <- tp / (tp + fn)
specificity      <- tn / (tn + fp)

# AUC Calculation
n_pos <- sum(test_preds$Duration_Category == "Over_2_Hours")
n_neg <- sum(test_preds$Duration_Category == "Under_2_Hours")
ranks <- rank(test_preds$prob_Over_2H, ties.method = "average")
auc_val <- (sum(ranks[test_preds$Duration_Category == "Over_2_Hours"]) - n_pos * (n_pos + 1) / 2) / (n_pos * n_neg)
